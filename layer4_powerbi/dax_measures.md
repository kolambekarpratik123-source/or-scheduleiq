```dax
-- ────────────────────────────────────────────────────────────
-- [Base] Total Cases
-- ────────────────────────────────────────────────────────────
[Total_Cases] =

-- ────────────────────────────────────────────────────────────
-- [Base] Completed Cases
-- ────────────────────────────────────────────────────────────
[Completed_Cases] =
CALCULATE(
    COUNTROWS( Fact_Surgical_Schedule ),
    Fact_Surgical_Schedule[CaseStatus] = "Completed"
)

-- ────────────────────────────────────────────────────────────
-- [Base] Cancelled Cases
-- ────────────────────────────────────────────────────────────
[Cancelled_Cases] =
CALCULATE(
    COUNTROWS( Fact_Surgical_Schedule ),
    Fact_Surgical_Schedule[CaseStatus] = "Cancelled"
)

-- ────────────────────────────────────────────────────────────
-- [Base] Emergency Cases
-- ────────────────────────────────────────────────────────────
[Emergency_Cases] =
CALCULATE(
    COUNTROWS( Fact_Surgical_Schedule ),
    Fact_Surgical_Schedule[EmergencyFlag] = TRUE()
)

-- ────────────────────────────────────────────────────────────
-- [Base] Total Scheduled Duration (hours)
-- ────────────────────────────────────────────────────────────
[Total_Scheduled_Hours] =
DIVIDE(
    SUMX(
        Fact_Surgical_Schedule,
        Fact_Surgical_Schedule[ScheduledDuration_min]

    60
)

-- ────────────────────────────────────────────────────────────
-- [Base] Total Actual Duration (hours)

[Total_Actual_Hours] =
DIVIDE(
    SUMX(
        Fact_Surgical_Schedule,
        Fact_Surgical_Schedule[ActualDuration_min]
    ),
    60
)
```

## SECTION 2: OR Utilization Measures

```dax
-- ────────────────────────────────────────────────────────────
-- [Utilization] OR_Utilization_%
-- Net utilization = Actual surgical time / Available OR time
-- Available = Rooms × Working Days × 10hr day
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[OR_Utilization_%] =
VAR TotalActualMin =
    SUMX(
        Fact_Surgical_Schedule,
        IF(
            Fact_Surgical_Schedule[CaseStatus] <> "Cancelled",
            Fact_Surgical_Schedule[ActualDuration_min],
            0
        )
    )
VAR DistinctDays =
    DISTINCTCOUNT( Fact_Surgical_Schedule[DateKey] )
VAR NumRooms =
    DISTINCTCOUNT( Fact_Surgical_Schedule[RoomID] )
VAR AvailableMin =
    NumRooms * DistinctDays * 600  -- 600 min = 10-hour OR day
VAR Result =
    DIVIDE( TotalActualMin, AvailableMin, 0 )
RETURN
    ROUND( Result, 4 )

-- ────────────────────────────────────────────────────────────
-- [Utilization] OR_Utilization_% (by Room)
-- Useful for bar chart breakdowns
-- ────────────────────────────────────────────────────────────
[OR_Utilization_%_ByRoom] =
VAR TotalActualMin =
    SUMX(
        Fact_Surgical_Schedule,
        IF(
            Fact_Surgical_Schedule[CaseStatus] <> "Cancelled",
            Fact_Surgical_Schedule[ActualDuration_min],
            0
        )
    )
VAR DistinctDays = DISTINCTCOUNT( Fact_Surgical_Schedule[DateKey] )
VAR AvailableMin = DistinctDays * 600  -- Single room context
RETURN
    ROUND( DIVIDE( TotalActualMin, AvailableMin, 0 ), 4 )

-- ────────────────────────────────────────────────────────────
-- [Utilization] Target_Utilization_Band
-- Returns label for conditional formatting
-- ────────────────────────────────────────────────────────────
[Target_Utilization_Band] =
VAR Util = [OR_Utilization_%]
RETURN
    SWITCH(
        TRUE(),
        Util >= 1.00, "🔴 OVERTIME",
        Util >= 0.85, "🟡 BUFFER BREACH",
        Util >= 0.70, "🟢 ON TRACK",
        "⚪ UNDERUTILIZED"
    )
```

## SECTION 3: Overtime Measures

```dax
-- ────────────────────────────────────────────────────────────
-- [Overtime] Total_Overtime_Minutes
-- Sum of minutes each case exceeded the OR day-end boundary
-- ────────────────────────────────────────────────────────────
[Total_Overtime_Minutes] =
SUMX(
    Fact_Surgical_Schedule,
    MAX( 0, Fact_Surgical_Schedule[OverrunMin] )
)

-- ────────────────────────────────────────────────────────────
-- [Overtime] Total_Overtime_Hours
-- Format String: "0.0 \h\r\s"
-- ────────────────────────────────────────────────────────────
[Total_Overtime_Hours] =
DIVIDE( [Total_Overtime_Minutes], 60 )

-- ────────────────────────────────────────────────────────────
-- [Overtime] Rooms_With_Overtime
-- Count of distinct rooms that ran past OR_END on any day
-- ────────────────────────────────────────────────────────────
[Rooms_With_Overtime] =
CALCULATE(
    DISTINCTCOUNT( Fact_Surgical_Schedule[RoomID] ),
    Fact_Surgical_Schedule[OverrunMin] > 0
)

-- ────────────────────────────────────────────────────────────
-- [Overtime] Projected_Overtime_Cost_INR  ★ KEY MEASURE
-- Responds to What-If sliders:
--   Buffer_Time_Minutes → adjusts planned case time
--   OT_CostRate_INR     → adjusts per-hour cost
--
-- Formula: OT_min_adjusted = max(0, actual - (scheduled + buffer))
--          Cost = (OT_min / 60) × rate
-- Format String: "₹ #,##0"
-- ────────────────────────────────────────────────────────────
[Projected_Overtime_Cost_INR] =
VAR BufferMin   = [Buffer_Time_Minutes Value]   -- From What-If slider
VAR CostPerHour = [OT_CostRate_INR Value]        -- From What-If slider
VAR OvertimeMinutes =
    SUMX(
        Fact_Surgical_Schedule,
        -- With the buffer, cases that barely overrun may be absorbed
        MAX(
            0,
            Fact_Surgical_Schedule[ActualDuration_min]
                - Fact_Surgical_Schedule[ScheduledDuration_min]
                - BufferMin
        )
    )
VAR OvertimeHours = DIVIDE( OvertimeMinutes, 60 )
RETURN
    ROUND( OvertimeHours * CostPerHour, 0 )

-- ────────────────────────────────────────────────────────────
-- [Overtime] OT_Cost_Comparison
-- Compares current buffer to 30-min baseline to show savings
-- ────────────────────────────────────────────────────────────
[OT_Cost_vs_Baseline_INR] =
VAR BaselineCost =
    CALCULATE(
        [Projected_Overtime_Cost_INR],
        'Buffer_Time_Minutes'[Buffer_Time_Minutes] = 30
    )
RETURN
    BaselineCost - [Projected_Overtime_Cost_INR]

-- ────────────────────────────────────────────────────────────
-- [Overtime] Overtime_Alert_Color
-- For KPI card conditional formatting
-- Returns hex color string
-- ────────────────────────────────────────────────────────────
[Overtime_Alert_Color] =
VAR OT = [Total_Overtime_Hours]
RETURN
    SWITCH(
        TRUE(),
        OT > 8,  "#DC2626",   -- Crimson Red: critical
        OT > 3,  "#D97706",   -- Amber: buffer breach
        OT > 0,  "#F59E0B",   -- Yellow: mild overrun
        "#059669"             -- Emerald: no overtime
    )
```

## SECTION 4: Case Cancellation Risk

```dax
-- ────────────────────────────────────────────────────────────
-- [Risk] Case_Cancellation_Rate_%
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[Case_Cancellation_Rate_%] =
DIVIDE(
    [Cancelled_Cases],
    [Total_Cases],
    0
)

-- ────────────────────────────────────────────────────────────
-- [Risk] Predicted_Cancellation_Risk_%
-- Uses historical complication + ASA data from Fact_Historical_Cases
-- Weighted composite score
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[Predicted_Cancellation_Risk_%] =
VAR AvgASA =
    AVERAGEX(
        Fact_Surgical_Schedule,
        Fact_Surgical_Schedule[ASA_Score]
    )
VAR EmergencyRate =
    DIVIDE( [Emergency_Cases], [Total_Cases], 0 )
VAR HighRiskRate =
    DIVIDE(
        CALCULATE(
            COUNTROWS( Fact_Surgical_Schedule ),
            Fact_Surgical_Schedule[RiskTierKey] >= 3  -- High / Critical
        ),
        [Total_Cases],
        0
    )
-- Logistic approximation based on published OR cancellation models
RETURN
    ROUND(
        0.03
        + 0.015 * (AvgASA - 2)
        + 0.12  * EmergencyRate
        + 0.08  * HighRiskRate,
        4
    )
```

```dax
-- ────────────────────────────────────────────────────────────
-- [PACU] Total_PACU_Holds
-- Cases exceeding 90-min PACU threshold (delayed room release)
-- ────────────────────────────────────────────────────────────
[Total_PACU_Holds] =
CALCULATE(
    COUNTROWS( Fact_Surgical_Schedule ),
    Fact_Surgical_Schedule[PACU_Hold_Flag] = TRUE()
)

-- ────────────────────────────────────────────────────────────
-- [PACU] PACU_Hold_Rate_%
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[PACU_Hold_Rate_%] =
DIVIDE( [Total_PACU_Holds], [Total_Cases], 0 )

-- ────────────────────────────────────────────────────────────
-- [PACU] Avg_PACU_Duration_min
-- Format String: "0 \m\i\n"
-- ────────────────────────────────────────────────────────────
[Avg_PACU_Duration_min] =
AVERAGEX(
    FILTER(
        Fact_Surgical_Schedule,
        NOT ISBLANK( Fact_Surgical_Schedule[PACU_Duration_min] )
    ),
    Fact_Surgical_Schedule[PACU_Duration_min]
)

-- ────────────────────────────────────────────────────────────
-- [PACU] PACU_Hold_Time_Hours
-- Total delayed minutes converted to hours
-- Format String: "0.0 \h\r\s"
-- ────────────────────────────────────────────────────────────
[PACU_Hold_Time_Hours] =
DIVIDE(
    SUMX(
        FILTER( Fact_Surgical_Schedule, Fact_Surgical_Schedule[PACU_Hold_Flag] = TRUE() ),
        MAX( 0, Fact_Surgical_Schedule[PACU_Duration_min] - 90 )
    ),
    60
)

-- ────────────────────────────────────────────────────────────
-- [PACU] PACU_Total_Hold_Cost_INR
-- Format String: "₹ #,##0"
-- ────────────────────────────────────────────────────────────
[PACU_Total_Hold_Cost_INR] =
SUMX(
    Fact_Surgical_Schedule,
    Fact_Surgical_Schedule[PACU_HoldCost_INR]
)
```

## SECTION 6: ICU / Downstream Bottleneck Measures

```dax
-- ────────────────────────────────────────────────────────────
-- [ICU] Total_ICU_Transfers
-- ────────────────────────────────────────────────────────────
[Total_ICU_Transfers] =
CALCULATE(
    COUNTROWS( Fact_Surgical_Schedule ),
    Fact_Surgical_Schedule[ICU_Transfer_Flag] = TRUE()
)

-- ────────────────────────────────────────────────────────────
-- [ICU] ICU_Transfer_Rate_%
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[ICU_Transfer_Rate_%] =
DIVIDE( [Total_ICU_Transfers], [Completed_Cases], 0 )

-- ────────────────────────────────────────────────────────────
-- [ICU] Avg_ICU_LOS_Hours
-- Format String: "0.0 \h\r\s"
-- ────────────────────────────────────────────────────────────
[Avg_ICU_LOS_Hours] =
AVERAGEX(
    FILTER(
        Fact_Surgical_Schedule,
        Fact_Surgical_Schedule[ICU_Transfer_Flag] = TRUE()
            && NOT ISBLANK( Fact_Surgical_Schedule[ICU_LOS_Hours] )
    ),
    Fact_Surgical_Schedule[ICU_LOS_Hours]
)

-- ────────────────────────────────────────────────────────────
-- [ICU] Peak_ICU_Bed_Demand  ★ KEY MEASURE
-- Estimates concurrent ICU occupancy based on LOS and arrival times
-- Uses the sim_hourly_bed_trace table (if connected)
-- Fallback: statistical approximation from fact table
-- Format String: "0 \B\e\d\s"
-- ────────────────────────────────────────────────────────────
[Peak_ICU_Bed_Demand] =
VAR ICUTransfers = [Total_ICU_Transfers]
VAR AvgLOS       = [Avg_ICU_LOS_Hours]
-- Little's Law: L = λ × W  (avg occupancy = arrival rate × avg stay)
-- Daily arrival rate = ICU_transfers / working_days
VAR WorkingDays  = DISTINCTCOUNT( Fact_Surgical_Schedule[DateKey] )
VAR DailyArrivals = DIVIDE( ICUTransfers, WorkingDays, 0 )
VAR AvgOccupancy = DailyArrivals * DIVIDE( AvgLOS, 24, 0 )  -- Convert hrs to days
-- Peak ≈ 1.5× average (based on queuing theory for Poisson arrivals)
RETURN
    ROUND( AvgOccupancy * 1.5, 0 )

-- ────────────────────────────────────────────────────────────
-- [ICU] ICU_Bed_Saturation_Risk  ★ KEY MEASURE
-- Percentage of ICU capacity consumed at peak demand
-- Format String: "0.0%"
-- Alert threshold: >80% = critical risk
-- ────────────────────────────────────────────────────────────
[ICU_Bed_Saturation_Risk] =
VAR PeakDemand  = [Peak_ICU_Bed_Demand]
VAR TotalBeds   = 16          -- From config: NUM_ICU_BEDS
VAR Saturation  = DIVIDE( PeakDemand, TotalBeds, 0 )
RETURN
    ROUND( Saturation, 4 )

-- ────────────────────────────────────────────────────────────
-- [ICU] ICU_Saturation_Label
-- For KPI card subtitle with traffic-light indicator
-- ────────────────────────────────────────────────────────────
[ICU_Saturation_Label] =
VAR Sat = [ICU_Bed_Saturation_Risk]
RETURN
    SWITCH(
        TRUE(),
        Sat >= 0.95, "🔴 CRITICAL — Immediate Action Required",
        Sat >= 0.80, "🟠 HIGH — ICU Block Risk Active",
        Sat >= 0.60, "🟡 MODERATE — Monitor Closely",
        "🟢 NORMAL — Within Safe Operating Range"
    )

-- ────────────────────────────────────────────────────────────
-- [ICU] ICU_Saturation_Color
-- Returns hex color for KPI card title color
-- ────────────────────────────────────────────────────────────
[ICU_Saturation_Color] =
VAR Sat = [ICU_Bed_Saturation_Risk]
RETURN
    SWITCH(
        TRUE(),
        Sat >= 0.95, "#DC2626",   -- Crimson
        Sat >= 0.80, "#EA580C",   -- Orange
        Sat >= 0.60, "#D97706",   -- Amber
        "#059669"                 -- Emerald
    )

-- ────────────────────────────────────────────────────────────
-- [ICU] Total_ICU_Bed_Cost_INR
-- Total cost of ICU stays across filtered period
-- Format String: "₹ #,##0"
-- ────────────────────────────────────────────────────────────
[Total_ICU_Bed_Cost_INR] =
SUMX(
    FILTER(
        Fact_Surgical_Schedule,
        Fact_Surgical_Schedule[ICU_Transfer_Flag] = TRUE()
    ),
    Fact_Surgical_Schedule[ICU_LOS_Hours] * 3500   -- ₹3,500/hour ICU cost
)
```

## SECTION 7: Dynamic Turnover Buffer (What-If Driven)

```dax
-- ────────────────────────────────────────────────────────────
-- [Buffer] Dynamic_Turnover_Buffer  ★ KEY MEASURE
-- Core What-If engine: recalculates daily schedule capacity
-- given the selected buffer time from the slider
-- Format String: "0 \m\i\n"
-- ────────────────────────────────────────────────────────────
[Dynamic_Turnover_Buffer] =
VAR BufferMin = [Buffer_Time_Minutes Value]
-- Return the selected buffer as a display measure
RETURN
    BufferMin

-- ────────────────────────────────────────────────────────────
-- [Buffer] Adjusted_Cases_Per_Room_Per_Day
-- How many cases fit in a 10-hr day with the chosen buffer
-- Assumes P50 procedure duration per specialty
-- ────────────────────────────────────────────────────────────
[Adjusted_Cases_Per_Room_Per_Day] =
VAR BufferMin      = [Buffer_Time_Minutes Value]
VAR AvgProcMin     = AVERAGEX(
                        Fact_Surgical_Schedule,
                        Fact_Surgical_Schedule[ScheduledDuration_min]
                    )
VAR SlotSize       = AvgProcMin + BufferMin
VAR CasesPerDay    = FLOOR( DIVIDE( 600, SlotSize, 0 ), 1 )
RETURN
    CasesPerDay

-- ────────────────────────────────────────────────────────────
-- [Buffer] Buffer_Efficiency_Rate_%
-- % of buffer time that is actually consumed by turnover
-- (shows whether buffer is too short/long)
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[Buffer_Efficiency_Rate_%] =
VAR BufferMin = [Buffer_Time_Minutes Value]
VAR AvgActualTurnover = AVERAGEX(
                            FILTER(
                                Fact_Surgical_Schedule,
                                NOT ISBLANK( Fact_Surgical_Schedule[TurnoverMin] )
                            ),
                            Fact_Surgical_Schedule[TurnoverMin]
                        )
RETURN
    ROUND( DIVIDE( AvgActualTurnover, BufferMin, 0 ), 4 )

-- ────────────────────────────────────────────────────────────
-- [Buffer] Wasted_Buffer_Hours
-- Hours of buffer time not consumed (opportunity cost)
-- Format String: "0.0 \h\r\s"
-- ────────────────────────────────────────────────────────────
[Wasted_Buffer_Hours] =
VAR BufferMin = [Buffer_Time_Minutes Value]
VAR TotalCases = [Total_Cases]
VAR AvgActualTurnover = AVERAGEX(
                            FILTER(
                                Fact_Surgical_Schedule,
                                NOT ISBLANK( Fact_Surgical_Schedule[TurnoverMin] )
                            ),
                            Fact_Surgical_Schedule[TurnoverMin]
                        )
VAR WastedMin = MAX( 0, BufferMin - AvgActualTurnover ) * TotalCases
RETURN
    ROUND( DIVIDE( WastedMin, 60 ), 1 )
```

## SECTION 8: Time Intelligence Measures

```dax
-- ────────────────────────────────────────────────────────────
-- [Time] Cases_MTD (Month-to-Date)
-- ────────────────────────────────────────────────────────────
[Cases_MTD] =
CALCULATE(
    [Total_Cases],
    DATESMTD( Dim_Date[Date] )
)

-- ────────────────────────────────────────────────────────────
-- [Time] OT_Cost_WoW_Change_%
-- Week-over-week change in overtime costs
-- Format String: "+0.0%;-0.0%;0.0%"
-- ────────────────────────────────────────────────────────────
[OT_Cost_WoW_Change_%] =
VAR CurrentWeekCost  = [Projected_Overtime_Cost_INR]
VAR PreviousWeekCost =
    CALCULATE(
        [Projected_Overtime_Cost_INR],
        DATEADD( Dim_Date[Date], -7, DAY )
    )
RETURN
    DIVIDE(
        CurrentWeekCost - PreviousWeekCost,
        NULLIF( PreviousWeekCost, 0 ),
        BLANK()
    )

-- ────────────────────────────────────────────────────────────
-- [Time] Rolling_7D_Avg_Utilization
-- 7-day rolling average utilization (smooth trend line)
-- ────────────────────────────────────────────────────────────
[Rolling_7D_Avg_Utilization] =
AVERAGEX(
    DATESINPERIOD(
        Dim_Date[Date],
        LASTDATE( Dim_Date[Date] ),
        -7,
        DAY
    ),
    [OR_Utilization_%]
)
```

## SECTION 9: Historical vs. Simulated Comparison

```dax
-- ────────────────────────────────────────────────────────────
-- [Compare] Hist_vs_Sim_Duration_Variance_%
-- How far does simulation deviate from historical actuals?
-- ────────────────────────────────────────────────────────────
[Hist_vs_Sim_Duration_Variance_%] =
VAR HistAvg =
    CALCULATE(
        AVERAGEX( Fact_Historical_Cases, Fact_Historical_Cases[ActualDuration_min] )
    )
VAR SimAvg =
    AVERAGEX( Fact_Surgical_Schedule, Fact_Surgical_Schedule[ActualDuration_min] )
RETURN
    DIVIDE( SimAvg - HistAvg, NULLIF( HistAvg, 0 ), BLANK() )

-- ────────────────────────────────────────────────────────────
-- [Compare] Surgeon_Velocity_Index_Avg
-- Average velocity index: <1 = faster, >1 = slower than scheduled
-- Format String: "0.00×"
-- ────────────────────────────────────────────────────────────
[Surgeon_Velocity_Index_Avg] =
AVERAGEX(
    FILTER(
        Fact_Historical_Cases,
        NOT ISBLANK( Fact_Historical_Cases[VelocityIndex] )
    ),
    Fact_Historical_Cases[VelocityIndex]
)

-- ────────────────────────────────────────────────────────────
-- [Compare] Exceeded_P90_Rate_%
-- % of cases that exceeded the 90th percentile benchmark
-- Format String: "0.0%"
-- ────────────────────────────────────────────────────────────
[Exceeded_P90_Rate_%] =
DIVIDE(
    CALCULATE(
        COUNTROWS( Fact_Historical_Cases ),
        Fact_Historical_Cases[ExceededP90_Flag] = TRUE()
    ),
    COUNTROWS( Fact_Historical_Cases ),
    0
)
```
